def analizar_columna(datos, nombre_columna):
    datos_limpios = datos.dropna().values
    if len(datos_limpios) == 0:
        return "La columna seleccionada no tiene datos numéricos válidos.", None

    media = np.mean(datos_limpios)
    mediana = np.median(datos_limpios)

    moda_res = stats.mode(datos_limpios, keepdims=True)
    moda = moda_res.mode[0] if len(moda_res.mode) > 0 else "N/A"

    desviacion_estandar = np.std(datos_limpios, ddof=1) if len(datos_limpios) > 1 else 0.0
    
    # CORREGIDO: Se une la línea que estaba cortada de forma incorrecta
    coeficiente_variacion = (desviacion_estandar / media) * 100 if media != 0 else 0.0

    q1 = np.percentile(datos_limpios, 25)
    q2 = np.percentile(datos_limpios, 50)
    q3 = np.percentile(datos_limpios, 75)

    # Análisis de Significancia Estadística (Diferencia de cero o Normalidad)
    if len(datos_limpios) > 1:
        stat_val, p_value = stats.ttest_1samp(datos_limpios, 0.0)
        shapiro_stat, shapiro_p = stats.shapiro(datos_limpios) if len(datos_limpios) < 5000 else (0, 0)
    else:
        p_value = 1.0
        shapiro_p = 1.0

    significativo = "Sí" if p_value < 0.05 else "No"
    es_normal = "Sí" if shapiro_p > 0.05 else "No"

    df_reporte = pd.DataFrame({
        'Métrica': [
            'Media (Promedio)',
            'Mediana (Q2)',
            'Moda',
            'Desviación Estándar',
            'Coeficiente de Variación',
            'Primer Cuartil (Q1)',
            'Tercer Cuartil (Q3)',
            'Total de Muestras',
            'P-Valor (Diferencia de 0)',
            '¿Es Significativo? (p < 0.05)',
            '¿Distribución Normal? (Shapiro-Wilk)'
        ],
        'Valor': [
            f"{media:.4f}",
            f"{mediana:.4f}",
            f"{moda}",
            f"{desviacion_estandar:.4f}",
            f"{coeficiente_variacion:.2f}%",
            f"{q1:.4f}",
            f"{q3:.4f}",
            f"{len(datos_limpios)}",
            f"{p_value:.5e}",
            f"{significativo}",
            f"{es_normal} (p-val: {shapiro_p:.4e})"
        ]
    })
    return df_reporte, datos_limpios

def al_subir_archivo(change):
    global df_loaded
    if not upload_btn.value:
        return

    # Manejo compatible de ipywidgets FileUpload según la versión
    val = upload_btn.value
    if isinstance(val, tuple):  # Para versiones recientes de ipywidgets
        archivo_info = val[0]
        nombre_archivo = archivo_info['name']
        contenido = archivo_info['content']
    else:  # Diccionario tradicional
        archivo_info = list(val.values())[0]
        contenido = archivo_info['content']
        nombre_archivo = archivo_info['metadata']['name']

    with out_config:
        clear_output()
        print(f"Archivo cargado exitosamente: {nombre_archivo}")

# Conectar evento de subida
upload_btn.observe(al_subir_archivo, names='value')
